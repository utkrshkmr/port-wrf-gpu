\chapter{Lateral boundaries and nesting}\label{ch:bdy}

How the outer domain is driven by boundary files, how the nest is driven by the outer domain, and the physical boundary conditions.

\begin{outline}[(card B-06)]
\textbf{Sections.}
\begin{itemize}
    \item \textbf{Specified and relaxation zones.} \code{spec_zone}, \code{relax_zone}; the boundary arrays and their tendencies; \code{relax_bdytend_core}, \code{spec_bdytend}, \code{spec_bdyupdate}.
    \item \textbf{One-way nesting.} \code{med_force_domain}: coupling, interpolation from the parent (\code{bdy_interp}), uncoupling; why couple-then-uncouple is not the identity in floating point.
    \item \textbf{Physical boundaries.} \code{set_physical_bc2d}, \code{set_physical_bc3d}; flow-dependent boundaries for scalars.
    \item \textbf{Time interpolation.} Boundary tendencies in time; the 3-hourly boundary file.
\end{itemize}
\textbf{Sources.} \citet{skamarock2019}.
